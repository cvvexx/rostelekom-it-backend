    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ProblemDetail> handleValidation(MethodArgumentNotValidException e, Locale locale) {
        String detail = messageSource.getMessage("error.validation-failed", null, locale);
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, detail);
        problem.setProperty("code", "error.validation-failed");

        Map<String, String> errors = e.getBindingResult().getFieldErrors().stream()
                .collect(Collectors.toMap(
                        FieldError::getField,
                        fe -> fe.getDefaultMessage() != null ? fe.getDefaultMessage() : "Invalid value",
                        (first, second) -> first
                ));
        problem.setProperty("errors", errors);
        return ResponseEntity.badRequest().body(problem);
    }

    @ExceptionHandler(AccessDeniedException.class)
    public ResponseEntity<ProblemDetail> handleAccessDenied(AccessDeniedException e, Locale locale) {
        String detail = messageSource.getMessage("error.forbidden", null, locale);
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.FORBIDDEN, detail);
        problem.setProperty("code", "error.forbidden");
        return ResponseEntity.status(HttpStatus.FORBIDDEN).body(problem);

import io.cvvexxx.services.subscription.dto.ChargeDto;
import io.cvvexxx.services.subscription.dto.CreateChargeDto;
import lombok.RequiredArgsConstructor;
import org.springframework.http.ProblemDetail;
import org.springframework.web.client.HttpClientErrorException;
import org.springframework.web.client.RestClient;

import java.util.Map;

@RequiredArgsConstructor
public class BillingServiceRestClientImpl implements BillingServiceRestClient {

    //Коды из messages.properties billing-service, приходят в поле code у ProblemDetail
    private static final String INSUFFICIENT_FUNDS_CODE = "charge.insufficient-funds";
    private static final String ACCOUNT_BLOCKED_CODE = "charge.account-blocked";
    private static final String ACCOUNT_NOT_FOUND_CODE = "account.not-found-by-user";
    private static final String ACCOUNT_CLOSED_CODE = "charge.account-closed";

    private final RestClient restClient;

    @Override
    public ChargeOutcome createCharge(CreateChargeDto createChargeDto) {
        try {
            ChargeDto charge = restClient
                    .post()
                    .uri("/internal/billing/charges")
                    .body(createChargeDto)
                    .retrieve()
                    .body(ChargeDto.class);
            return new ChargeOutcome.Charged(charge.transactionId());
        } catch (HttpClientErrorException.Conflict | HttpClientErrorException.NotFound e) {
            String code = problemCode(e);
            return switch (code) {
                case INSUFFICIENT_FUNDS_CODE, ACCOUNT_BLOCKED_CODE, ACCOUNT_NOT_FOUND_CODE -> new ChargeOutcome.Declined(code);
                case ACCOUNT_CLOSED_CODE -> new ChargeOutcome.AccountClosed();
                case null, default -> throw e;
            };
        }
    }

    private static String problemCode(HttpClientErrorException e) {
        ProblemDetail problem = e.getResponseBodyAs(ProblemDetail.class);
        Map<String, Object> properties = problem == null ? null : problem.getProperties();
        return properties != null && properties.get("code") instanceof String code ? code : null;
    }
}

package io.cvvexxx.services.client;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpRequest;
import org.springframework.http.HttpStatus;
import org.springframework.http.client.ClientHttpRequestExecution;
import org.springframework.http.client.ClientHttpRequestInterceptor;
import org.springframework.http.client.ClientHttpResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;

import java.io.IOException;
import java.time.Duration;
import java.time.Instant;
import java.util.concurrent.locks.ReentrantLock;

//Добавляет сервисный токен от user-service в межсервисные запросы (billing и т.п.).
//Токен держим в памяти и берём новый за REFRESH_BEFORE до истечения, refresh-токена у сервисного нет
@Component
public class ServiceTokenInterceptor implements ClientHttpRequestInterceptor {

    //Запас, чтобы токен не истёк, пока запрос идёт по сети
    private static final Duration REFRESH_BEFORE = Duration.ofSeconds(30);

    private final RestClient userServiceClient;
    private final ServiceTokenRequest credentials;

    private final ReentrantLock lock = new ReentrantLock();
    private volatile CachedToken token;

    public ServiceTokenInterceptor(RestClient.Builder builder,
                                   @Value("${rostelekom.uri.user-service}") String userServiceUri,
                                   @Value("${app.service-auth.client-id}") String clientId,
                                   @Value("${app.service-auth.client-secret}") String clientSecret) {
        this.userServiceClient = builder.baseUrl(userServiceUri).build();
        this.credentials = new ServiceTokenRequest(clientId, clientSecret);
    }

    @Override
    public ClientHttpResponse intercept(
            HttpRequest request, byte[] body, ClientHttpRequestExecution execution
    ) throws IOException {
        request.getHeaders().setBearerAuth(token());
        ClientHttpResponse response = execution.execute(request, body);
        if (response.getStatusCode().value() == HttpStatus.UNAUTHORIZED.value()) {
            token = null;
        }
        return response;
    }

    private String token() {
        CachedToken current = token;
        if (current != null && Instant.now().isBefore(current.refreshAt())) {
            return current.value();
        }
        lock.lock();
        try {
            //Пока ждали lock, токен мог обновить другой поток
            current = token;
            if (current != null && Instant.now().isBefore(current.refreshAt())) {
                return current.value();
            }
            ServiceTokenResponse response = userServiceClient
                    .post()
                    .uri("/internal/auth/service-token")
                    .body(credentials)
                    .retrieve()
                    .body(ServiceTokenResponse.class);
            if (response == null || response.accessToken() == null) {
                throw new IllegalStateException("user-service вернул пустой сервисный токен");
            }
            Instant refreshAt = Instant.now().plusSeconds(response.expiresInSeconds()).minus(REFRESH_BEFORE);
            token = new CachedToken(response.accessToken(), refreshAt);
            return response.accessToken();
        } finally {
            lock.unlock();
        }
    }

    private record CachedToken(String value, Instant refreshAt) {
    }

    private record ServiceTokenRequest(String clientId, String clientSecret) {
    }

    private record ServiceTokenResponse(String accessToken, long expiresInSeconds) {
    }
}

package io.cvvexxx.services.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationConverter;
import org.springframework.security.web.SecurityFilterChain;

import java.util.List;
import java.util.stream.Collectors;

@Configuration
public class SecurityBeans {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        return http
                .csrf(AbstractHttpConfigurer::disable)
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/v3/api-docs/", "/swagger-ui/", "/swagger-ui.html").permitAll()
                        .requestMatchers("/api/services/tariffs/all").hasRole("ADMIN")
                        .requestMatchers(HttpMethod.GET, "/api/services/tariffs", "/api/services/tariffs/*").permitAll()
                        .anyRequest().authenticated()
                )
                .oauth2ResourceServer(oauth2 -> oauth2
                        .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter()))
                )
                .build();
    }

    @Bean
    public JwtAuthenticationConverter jwtAuthenticationConverter() {
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(jwt -> {
            List<String> roles = jwt.getClaimAsStringList("roles");
            if (roles == null) {
                return List.of();
            }
            return roles.stream()
                    .map(role -> role.startsWith("ROLE_") ? role : "ROLE_" + role)
                    .map(SimpleGrantedAuthority::new)
                    .collect(Collectors.toList());
        });
        return converter;
    }
}

            @AuthenticationPrincipal Jwt jwt,

            @PathVariable("id") UUID id
    ) {

    }

import io.cvvexxx.services.client.BillingServiceRestClient;
import io.cvvexxx.services.client.ChargeOutcome;

import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.scheduling.annotation.Scheduled;

import java.time.Duration;
import java.time.Instant;
import java.time.ZoneOffset;
import java.util.List;
import java.util.UUID;

@Component
@RequiredArgsConstructor
@Slf4j
public class SubscriptionChargingJob {
    private static final int BATCH_SIZE = 100;

    @Value("${app.subscription.lease-ttl:2000}")
    private long leaseTtl;

    //Интервал ретраев после отказа billing растёт вместе с просрочкой: от min до max
    @Value("${app.subscription.charge-retry-min-delay:PT1H}")
    private Duration minRetryDelay;

    @Value("${app.subscription.charge-retry-max-delay:PT24H}")
    private Duration maxRetryDelay;

    //Сколько приостановленная подписка ждёт оплату, потом отменяем её и больше не ходим в billing
    @Value("${app.subscription.grace-period:P30D}")
    private Duration gracePeriod;

    private final BillingServiceRestClient restClient;
    private final SubscriptionRepository subscriptionRepository;
    private final SubscriptionChargeService subscriptionChargeService;


    @Scheduled(fixedDelayString = "${app.subscription.charge-interval:PT1M}")
    public void chargeDueSubscriptions() {
        List<DueSubscription> batch;
        do {
            Instant now = Instant.now();
            //Берем подписки которые скоро нужно оплатить
            batch = subscriptionRepository.claimDueForCharge(now, now.plus(Duration.ofMillis(leaseTtl)), BATCH_SIZE);
            for (DueSubscription s : batch) {
                try {
                    //По одной пытаемся обработать
                    process(s, now);
                } catch (Exception e) {
                    log.error("Не удалось списать по подписке {}", s.getId(), e);
                }
            }
        } while (batch.size() == BATCH_SIZE);
    }


    private void process(DueSubscription subscription, Instant now) {
        UUID id = subscription.getId();
        Instant claimedNextChargeAt = subscription.getNextChargeAt();

        if (subscription.getCancelAtPeriodEnd()) {
            subscriptionChargeService.markCanceled(id, claimedNextChargeAt, now);
            return;
        }

        //Проверка сколько уже не оплачена подписка(now - nextChargeAt), если больше gracePeriod то отменяем
        //Чтобы не ходить в billing service и не нагружать его просто так
        //TODO(В будущем добавить эндпоинт/кафку чтобы при пополнении баланса сервис услуг пытался списать деньги)
        Duration overdue = Duration.between(claimedNextChargeAt, now);
        if (subscription.getStatus() == SubscriptionStatus.SUSPENDED && overdue.compareTo(gracePeriod) > 0) {
            log.info("Подписка {} не оплачена дольше {}, отменяем", id, gracePeriod);
            subscriptionChargeService.markCanceled(id, claimedNextChargeAt, now);
            return;
        }

        ChargeOutcome outcome = restClient.createCharge(new CreateChargeDto(
                subscription.getUserId(),
                subscription.getPrice(),
                idempotencyKey(subscription),
                "Оплата тарифа " + subscription.getTariffName()
        ));

        switch (outcome) {
            case ChargeOutcome.Charged charged -> {
                //Приостановленную подписку оплатили только сейчас: новый период считаем от оплаты, а не от пропущенной даты
                Instant periodStart = subscription.getStatus() == SubscriptionStatus.SUSPENDED ? now : claimedNextChargeAt;
                //TODO(исправить если появятся подписки у которых оплата не раз в месяц)
                Instant nextChargeAt = periodStart.atOffset(ZoneOffset.UTC).plusMonths(1).toInstant();
                if (
                        !subscriptionChargeService.markCharged(id, claimedNextChargeAt, periodStart, nextChargeAt,
                                charged.transactionId())
                ) {
                    //подписку могли отменить пока мы пытались списать деньги, по факту пользователь
                    //заплатил за период отмененной подписки
                    //TODO(сделать возврат через billing, когда появится refund для charge)
                    log.warn("Списали {} по подписке {}, но результат не применён: подписку отменили или уже продлили. Транзакция {}",
                            subscription.getPrice(), id, charged.transactionId());
                }
            }
            //Не хватило денег, аккаунт не найден/заблокирован
            case ChargeOutcome.Declined declined -> {
                log.info("billing отказал в списании по подписке {}: {}", id, declined.code());
                //Растущий delay чтобы часто не нагружать billing service
                Duration retryDelay = Duration.ofSeconds(
                        Math.clamp(overdue.toSeconds(), minRetryDelay.toSeconds(), maxRetryDelay.toSeconds()));
                subscriptionChargeService.markSuspended(id, claimedNextChargeAt, now.plus(retryDelay));
            }
            //Аккаунт закрыт -> отменяем подписку
            case ChargeOutcome.AccountClosed ignored ->
                    subscriptionChargeService.markCanceled(id, claimedNextChargeAt, now);
        }
    }


    private static String idempotencyKey(DueSubscription subscription) {
        return subscription.getId() + ":" + subscription.getNextChargeAt().getEpochSecond();
    }

    @Transactional
    public boolean markCharged(UUID id, Instant claimedNextChargeAt, Instant periodStart, Instant nextChargeAt,
                               UUID transactionId) {

            subscription.setStatus(SubscriptionStatus.ACTIVE);
            subscription.setCurrentPeriodStart(periodStart);
            subscription.setNextChargeAt(nextChargeAt);

            subscription.setChargeLockedUntil(null);
            return true;
        }).orElse(false);
    }

    @Transactional
    public boolean markSuspended(UUID id, Instant claimedNextChargeAt, Instant retryAt) {
        return findClaimed(id, claimedNextChargeAt).map(subscription -> {
            subscription.setStatus(SubscriptionStatus.SUSPENDED);
            subscription.setChargeLockedUntil(retryAt);
            return true;
        }).orElse(false);
    }

    @Transactional
    public boolean markCanceled(UUID id, Instant claimedNextChargeAt, Instant now) {
        return findClaimed(id, claimedNextChargeAt).map(subscription -> {
            subscription.setStatus(SubscriptionStatus.CANCELED);
            subscription.setCanceledAt(now);
            subscription.setChargeLockedUntil(null);
            return true;
        }).orElse(false);
    }

    //Подписка всё ещё та, что взяли в claim: не отменена и next_charge_at не сдвинули
    private Optional<Subscription> findClaimed(UUID id, Instant claimedNextChargeAt) {

    public SubscriptionDto subscribe(UUID userId, UUID tariffId) {
        UUID subscriptionId = UUID.randomUUID();
        Instant now = Instant.now().truncatedTo(ChronoUnit.MICROS);
        Instant nextChargeAt = now.atOffset(ZoneOffset.UTC).plusMonths(1).toInstant();
        Instant lockedUntil = now.plus(Duration.ofMillis(leaseTtl));

        //резервация активной подписки в БД
        Tariff tariff = transactionTemplate.execute(status -> {
            Tariff found = tariffRepository.findById(tariffId)
                    .orElseThrow(() -> new TariffNotFoundException("tariff.not-found", tariffId));
            if (!found.isActive()) {
                throw new TariffInactiveException("tariff.inactive", tariffId);
            }

            int inserted = subscriptionRepository.insertActiveIfAbsent(
                    subscriptionId,
                    userId,
                    tariffId,
                    found.getPrice(),
                    found.getCurrency(),
                    now,
                    nextChargeAt,
                    lockedUntil
            );
            if (inserted == 0) {
                throw new SubscriptionAlreadyExistsException("subscription.already-exists", tariffId);
            }
            return found;
        });

        //списание
        ChargeOutcome outcome;
        try {
            outcome = billingClient.createCharge(new CreateChargeDto(
                    userId,
                    tariff.getPrice(),
                    subscriptionId + ":" + now.getEpochSecond(),
                    "Оплата тарифа " + tariff.getName()
            ));
        } catch (RuntimeException e) {
            transactionTemplate.executeWithoutResult(status -> subscriptionRepository.deleteById(subscriptionId));
            throw e;
        }

        //сохраняем транзакцию или удаляем черновик при отказе
        return switch (outcome) {
            case ChargeOutcome.Charged charged -> transactionTemplate.execute(status -> {
                Subscription subscription = subscriptionRepository.findByIdForUpdate(subscriptionId)
                        .orElseThrow(() -> new SubscriptionNotFoundException("subscription.not-found", subscriptionId));
                subscription.setLastChargeTransactionId(charged.transactionId());
                subscription.setChargeLockedUntil(null);
                return toDto(subscription);
            });
            case ChargeOutcome.Declined declined -> {
                transactionTemplate.executeWithoutResult(status -> subscriptionRepository.deleteById(subscriptionId));
                throw new ChargeDeclinedException(declined.code(), userId);
            }
            case ChargeOutcome.AccountClosed ignored -> {
                transactionTemplate.executeWithoutResult(status -> subscriptionRepository.deleteById(subscriptionId));
                throw new ChargeDeclinedException("charge.account-closed", userId);
            }
        };
    }

        return toDto(subscription);

        Subscription subscription = (isAdmin
                ? subscriptionRepository.findByIdForUpdate(subscriptionId)
                : subscriptionRepository.findByIdAndUserIdForUpdate(subscriptionId, userId))
                .orElseThrow(() -> new SubscriptionNotFoundException("subscription.not-found", subscriptionId));

        if (subscription.getStatus() == SubscriptionStatus.CANCELED) {
            throw new SubscriptionAlreadyCanceledException("subscription.already-canceled", subscriptionId);
        }

        if (immediate || subscription.getStatus() == SubscriptionStatus.SUSPENDED) {
            subscription.setStatus(SubscriptionStatus.CANCELED);
            subscription.setCanceledAt(Instant.now());
            subscription.setCancelAtPeriodEnd(false);
            subscription.setChargeLockedUntil(null);

            subscription.setCancelAtPeriodEnd(true);
        }

    }

                s.getCreatedAt()

package io.cvvexxx.services.tariff.service;

import io.cvvexxx.services.tariff.dto.CreateTariffDto;
import io.cvvexxx.services.tariff.dto.TariffDto;
import io.cvvexxx.services.tariff.dto.UpdateTariffDto;

import io.cvvexxx.services.tariff.entity.Tariff;
import io.cvvexxx.services.tariff.entity.TariffCategory;
import io.cvvexxx.services.tariff.exception.TariffAlreadyExistsException;
import io.cvvexxx.services.tariff.exception.TariffNotFoundException;
import io.cvvexxx.services.tariff.repository.TariffRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.UUID;

@Service
@RequiredArgsConstructor
@Slf4j
public class TariffService {

    private final TariffRepository tariffRepository;

    @Transactional(readOnly = true)
    public Page<TariffDto> getActiveTariffs(TariffCategory category, Pageable pageable) {
        Page<Tariff> tariffs = category == null
                ? tariffRepository.findByActiveTrue(pageable)
                : tariffRepository.findByActiveTrueAndCategory(category, pageable);

        return tariffs.map(this::toDto);
    }

    @Transactional(readOnly = true)
    public Page<TariffDto> getAllTariffs(TariffCategory category, Pageable pageable) {
        Page<Tariff> tariffs = category == null
                ? tariffRepository.findAll(pageable)
                : tariffRepository.findByCategory(category, pageable);

        return tariffs.map(this::toDto);
    }

    @Transactional(readOnly = true)
    public TariffDto getTariffById(UUID id, boolean isAdmin) {
        Tariff tariff = (isAdmin
                ? tariffRepository.findById(id)
                : tariffRepository.findByIdAndActiveTrue(id))
                .orElseThrow(() -> new TariffNotFoundException("tariff.not-found", id));
        return toDto(tariff);
    }

    @Transactional
    public TariffDto createTariff(CreateTariffDto dto) {
        String name = dto.name().trim();
        String description = dto.description() != null ? dto.description().trim() : null;
        BillingPeriod period = dto.billingPeriod() != null ? dto.billingPeriod() : BillingPeriod.MONTH;

        if (tariffRepository.insertIfAbsent(name, description, dto.category().name(), dto.price(), period.name()) == 0) {
            throw new TariffAlreadyExistsException("tariff.already-exists", name);
        }

        Tariff saved = tariffRepository.findByName(name)
                .orElseThrow(() -> new TariffNotFoundException("tariff.not-found-by-name", name));
        return toDto(saved);
    }

    @Transactional
    public TariffDto updateTariff(UUID id, UpdateTariffDto dto) {
        Tariff tariff = findTariffForUpdate(id);

        if (dto.name() != null) {
            String newName = dto.name().trim();
            if (tariffRepository.existsByNameAndIdNot(newName, id)) {
                throw new TariffAlreadyExistsException("tariff.already-exists", newName);
            }
            tariff.setName(newName);
        }
        if (dto.description() != null) {
            tariff.setDescription(dto.description().trim());
        }
        if (dto.category() != null) {
            tariff.setCategory(dto.category());
        }
        if (dto.price() != null) {
            tariff.setPrice(dto.price());
        }
        if (dto.active() != null) {
            tariff.setActive(dto.active());
        }

        return toDto(tariff);
    }

    @Transactional
    public TariffDto archiveTariff(UUID id) {
        Tariff tariff = findTariffForUpdate(id);
        tariff.setActive(false);
        return toDto(tariff);
    }

    private Tariff findTariffForUpdate(UUID id) {
        return tariffRepository.findByIdForUpdate(id)
                .orElseThrow(() -> new TariffNotFoundException("tariff.not-found", id));
    }

    private TariffDto toDto(Tariff tariff) {
        return new TariffDto(
                tariff.getId(),
                tariff.getName(),
                tariff.getDescription(),
                tariff.getCategory(),
                tariff.getPrice(),
                tariff.getCurrency(),
                tariff.getBillingPeriod(),
                tariff.isActive(),
                tariff.getCreatedAt()
        );
    }
}

        return jdbc.queryForObject("""

                returning id

                Timestamp.from(nextChargeAt.minus(Duration.ofDays(30))), Timestamp.from(nextChargeAt),
                chargeLockedUntil == null ? null : Timestamp.from(chargeLockedUntil));
    }

        Timestamp value = jdbc.queryForObject("select charge_locked_until from subscriptions where id = ?", Timestamp.class, id);
        return value == null ? null : value.toInstant();
